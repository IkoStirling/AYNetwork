if(TARGET AYNetworkHttp)
    return()
endif()
find_package(httplib CONFIG REQUIRED)
add_library(AYNetworkHttp STATIC src/HttpClient.cpp src/SystemProxy.cpp)
target_include_directories(AYNetworkHttp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(AYNetworkHttp PUBLIC cxx_std_23)
target_link_libraries(AYNetworkHttp PRIVATE httplib::httplib)
if(WIN32)
    target_link_libraries(AYNetworkHttp PRIVATE winhttp)
    target_compile_definitions(AYNetworkHttp PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(BUILD_TESTING AND TARGET AYTest)
    add_executable(AYNetwork_HttpTests unittest/Test_HttpClient.cpp)
    target_link_libraries(AYNetwork_HttpTests PRIVATE AYNetworkHttp AYTest httplib::httplib)
    ay_add_test(AYNetwork_HttpTests CASE_REPORT TIMEOUT 30 LABELS "network;services;unit")
endif()
